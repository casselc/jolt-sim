type Stations = { id : int; code : string; name : option(string<max 20>); kind : option(string); elevation : option(double); staffed : option(bool); ref : option(string); note : option(string); };

cell check_Stations(x : Stations) : bool = {
  ((!((type(x.id) == double || type(x.id) == int || type(x.id) == uint)) || x.id >= 1.0) && ((!(type(x.code) == string) || size(x.code) >= 1) && (!(type(x.code) == string) || x.code.matches("^(?:[A-Z]{3})$"))) && (!x.name.hasValue() || ((!(type(x.name.value()) == string) || size(x.name.value()) >= 1) && (!(type(x.name.value()) == string) || size(x.name.value()) <= 20))) && (!x.kind.hasValue() || (x.kind.value() in [dyn("metro"), dyn("rail"), dyn("bus")] && (!(type(x.kind.value()) == string) || size(x.kind.value()) >= 1))) && (!x.elevation.hasValue() || ((!((type(x.elevation.value()) == double || type(x.elevation.value()) == int || type(x.elevation.value()) == uint)) || x.elevation.value() >= -500.0) && (!((type(x.elevation.value()) == double || type(x.elevation.value()) == int || type(x.elevation.value()) == uint)) || x.elevation.value() <= 9000.0))) && (!x.ref.hasValue() || ((!(type(x.ref.value()) == string) || x.ref.value().matches("^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$")) && (!(type(x.ref.value()) == string) || size(x.ref.value()) >= 1))));
}
